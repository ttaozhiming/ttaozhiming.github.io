【题解】[SNOI2019] 通信

<!-- MORE -->

### 题目描述：

$n$ 个排成一列的哨站要进行通信。第 $i$ 个哨站的频段为 $a_i$。

每个哨站 $i$ 需要选择以下二者之一：

1. 直接连接到控制中心，代价为 $W$；
2. 连接到前面的某个哨站 $j$ ($j<i$)，代价为 $|a_i-a_j|$。
   每个哨站只能被后面的至多一个哨站连接。

请你求出最小可能的代价和。

对于所有数据，$1 \leq n \leq 1000$，$0 \leq W,a_i \leq 10^9$。

### 题目分析：

首先看到至多一个，想到可不可以二分图，发现根本不是二分图的形式，那就用网络流去做，通过题目性质可知，需要用到费用流。

对于一个 $i\to j$，其实就是 $i$ 向 $j$ 连一条权值为 $|a_i-a_j|$，费用为 $1$ 的边。

因为原点连向 $i$，$i$ 还需要连向汇点，并且流量都是 $1$，那么如果不把 $i$ 拆点，那就寄了。

对于 $i\to j$，就变成了 $i$ 的入点，连向 $j$ 的出点。 

这样就可以得到 $80$ 分的高分。

考虑怎么优化建图，绝对值非常难受，把他改成小于等于 $a_i$ 的和大于 $a_i$ 的，先看小于等于的，大于同理。

那么对于一个连边需要保证以下性质：$i\le j  \And  a_i\le a_j$。

是个二维偏序的形式，用分治优化。

const int INF=1e18+7;
const int N=1e5+7,M=1e7+7;

int n,W,tot,S,T,head[M],cnt=1;
int a[N],f[M],cur[N],dis[N],incf[N],lst[N];
int pos[N],b[N],pre[N],suf[N],in[N],out[N];
int tmp[N];
bool vis[N];

struct node{
    int v,nxt,w;
}e[M];

void add(int u,int v,int c,int d)
{
	e[++cnt].v=v;e[cnt].w=d;f[cnt]=c;e[cnt].nxt=head[u];head[u]=cnt;
	e[++cnt].v=u;e[cnt].w=-d;f[cnt]=0;e[cnt].nxt=head[v];head[v]=cnt;
}

bool spfa()
{
	queue<int> q;
	memset(dis,0x3f,sizeof dis);
	memset(incf,0,sizeof incf);
	q.push(S);incf[S]=INF;dis[S]=0;
	while(!q.empty())
	{
		int u=q.front();q.pop();
		vis[u]=0;
		for (int i=head[u];i;i=e[i].nxt)
		{
			int v=e[i].v;
			if (f[i]&&dis[v]>dis[u]+e[i].w)
			{
				dis[v]=dis[u]+e[i].w;
				lst[v]=i;
				incf[v]=min(incf[u],f[i]);
				if (!vis[v])
				{
					q.push(v);
					vis[v]=1;
				}
			}
		}
	}
	return incf[T]>0;
}

int EK()
{
	int flow=0,cost=0;
	while(spfa())
	{
		int t=incf[T];
		flow+=t;cost+=t*dis[T];
		for (int i=T;i!=S;i=e[lst[i]^1].v)
		{
			f[lst[i]]-=t;
			f[lst[i]^1]+=t;
		}
	}
    return cost;
}

int find(int l,int r,int val){
    int ans=r+1;
    while(l<=r){
        int mid=(l+r)>>1;
        if (a[b[mid]]>val) ans=mid,r=mid-1;
        else l=mid+1;
    }
    return ans;
}

void solve(int l,int r){
    if (l>=r) return;
    int mid=(l+r)>>1;
    solve(l,mid),solve(mid+1,r);

    for (int i=l;i<=mid;i++){
        pre[i]=++tot;add(pre[i],out[b[i]],1,-a[b[i]]);
        if (i!=l) add(pre[i],pre[i-1],INF,0);
    }
    for (int i=mid;i>=l;i--){
        suf[i]=++tot;add(suf[i],out[b[i]],1,a[b[i]]);
        if (i!=mid) add(suf[i],suf[i+1],INF,0);
    }

    for (int i=mid+1;i<=r;i++){
        int pos=find(l,mid,a[b[i]]);//找到第一个大于 a[b[i]] 的地方
        if (pos-1>=l) add(in[b[i]],pre[pos-1],1,a[b[i]]);
        if (pos<=mid) add(in[b[i]],suf[pos],1,-a[b[i]]);
    }

    int posl=l,posr=mid+1;int idx=l-1;
    while(posl<=mid&&posr<=r){
        if (a[b[posl]]<=a[b[posr]]) tmp[++idx]=b[posl],posl++;
        else tmp[++idx]=b[posr],posr++;
    }
    while(posl<=mid) tmp[++idx]=b[posl],posl++;
    while(posr<=r) tmp[++idx]=b[posr],posr++;
    for (int i=l;i<=r;i++) b[i]=tmp[i];
}

signed main(){
    scanf("%lld%lld",&n,&W);
    S=2*n+2,T=2*n+3;int s=2*n+4;tot=2*n+5;
    for (int i=1;i<=n;i++) scanf("%lld",&a[i]);
    add(s,T,INF,0);
    for (int i=1;i<=n;i++) in[i]=i*2,out[i]=i*2+1;
    for (int i=1;i<=n;i++) {
        add(S,in[i],1,0);
        add(out[i],T,1,0);
        add(in[i],s,1,W);
    }
    for (int i=1;i<=n;i++) b[i]=i;
    solve(1,n);
    printf("%lld\n",EK());
    return 0;
}
